{{- /* Page-bundle screenshot as WebP (max 1600 px wide, plus an 800 px variant), lazy-loaded, with width/height. Not linked: the image can't be opened on its own. Usage: {{< img src="payment-queue.png" alt="..." caption="..." >}} A missing file shows a placeholder in development and fails the production build. */ -}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" -}} {{- $caption := .Get "caption" -}} {{- if not $src }}{{ errorf "%s: img needs src" .Position }}{{ end -}} {{- if not $alt }}{{ errorf "%s: img %q needs alt text" .Position $src }}{{ end -}} {{- with .Page.Resources.Get $src -}} {{- $w := cond (gt .Width 1600) 1600 .Width -}} {{- $full := .Resize (printf "%dx webp q75" $w) -}} {{- $srcset := printf "%s %dw" $full.RelPermalink $full.Width -}} {{- if ge $w 1200 -}} {{- $half := .Resize "800x webp q75" -}} {{- $srcset = printf "%s %dw, %s" $half.RelPermalink $half.Width $srcset -}} {{- end }} {{- else -}} {{- if hugo.IsProduction -}} {{- errorf "%s: missing image %q in the page bundle" .Position $src -}} {{- else }}
Missing image: {{ $src }}
{{- end -}} {{- end -}}
